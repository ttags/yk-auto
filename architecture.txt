Full FLOW:
- APi scraper.js scrapes the match data from the websites ie. helytv, yono, yosin, hora, at the 60 seconds after match start time. ( to let the timegated match links propagate well ) 
- Run liveness checker and auto rank the links from best to worst ( based on the score of course) 
- Extracted data is formatted to apiauto.json ( match basic info only) and manifestauto.txt (encrypted stream links) 
- manifestauto is encrypted with custom encryption, not web crypto to avoid hooks
- Both file ie. encrypted manifestauto.txt and unencrypted apiauto.json are pushed to Github. 

- At frontend homepage, the match list is fetched from apiauto.json and shown. 
- If matches have Ended, they are hidden automatically from homepage. 
- Clicking match takes to the detail page with pillar match related article passing id= parameter. 

- Detail page takes id parameter and again fetches the apiauto.json and finds which entry has that id, 
        - if id is wrong, missing or invalid, nothing is shown.  
        - if id is right, it checks time in apiauto.json (main sauce with links ie. manifestauto.txt is not fetched yet)
                        - if time is more than 30 minutes --> Show message ie. Check back 30 minutes before match time for the real time scores. 
                        - if match is under 30 minutes --> Show a beautiful countdown timer. 
                        - NOTE: The 30 MINUTE countdown HAS 3 Extra minutes, that will keep ticking past the real start time to compensate for time taken to find and update links

        - If countdown finished, reveal the button - ie. Watch live ( no fetching of manifestauto.txt yet)
        - Clicking watch live, calls the fetch function to manifestauto.txt
        - It is then decrypted using custom decryption to form the match links and populated as blue pill buttons. 
        - If manifestauto.txt has no links, malformed content, invalid code, error is handled gracefully.

Files setup:
apiauto.json - plain match data. 
manifestauto.json - includes the todays current match window links in encrypted form.

<script>
const JSON_URL = "https://yk-auto.pages.dev/apiauto.json";

// Double-fetch guard
let refreshPending = false;

function convertToLocalTime(isoTime) {
  const date = new Date(isoTime);
  const timeString = date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', hour12: true });
  const dateString = date.toLocaleDateString('en-CA');
  return { time: timeString, date: dateString };
}

function getTimeDifference(startTime) {
  const now = new Date();
  const start = new Date(startTime);
  const diffMs = start - now;
  if (diffMs <= 0) return null;

  const diffDays    = Math.floor(diffMs / (1000 * 60 * 60 * 24));
  const diffHours   = Math.floor((diffMs % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
  const diffMinutes = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60));
  const diffSeconds = Math.floor((diffMs % (1000 * 60)) / 1000);

  if (diffDays > 0)   return `Starts in ${diffDays}d ${diffHours}h`;
  if (diffHours >= 1) return `Starts in ${diffHours}h ${diffMinutes}m`;
  return `Kicking off in ${diffMinutes}m:${diffSeconds.toString().padStart(2, '0')}s`;
}

function getMatchStatus(match) {
  const now   = new Date();
  const start = new Date(match.start_time);
  const end   = new Date(match.end_time);
  if (now >= start && now < end) return 'live';
  if (now >= end) return 'ended';
  return 'upcoming';
}

function getMatchUrl(match) {
  return `https://www.yonotvs.com/p/todays-match-details.html?id=${match.id || ''}`;
}

function createMatchCard(match) {
  const status        = getMatchStatus(match);
  const localTime     = convertToLocalTime(match.start_time);
  const timeRemaining = getTimeDifference(match.start_time);

  let statusContent = '';
  let footerContent = '';

  if (status === 'live') {
    statusContent = '<div class="live-indicator">LIVE</div>';
    footerContent = '<div class="live-score-btn">Live Now \u2192</div>';
  } else if (status === 'ended') {
    statusContent = `
      <div class="match-time">END</div>
      <div class="match-date">${localTime.date}</div>
    `;
    footerContent = '<div class="live-score-btn">Match Finished</div>';
  } else {
    statusContent = `
      <div class="match-time">${localTime.time}</div>
      <div class="match-date">${localTime.date}</div>
    `;
    footerContent = `<div class="live-score-btn">${timeRemaining || 'Kicking off soon'} \u2192</div>`;
  }

  const badgeText = status === 'live' ? 'LIVE'
                  : status === 'upcoming' ? 'UPCOMING'
                  : 'ENDED';

  const homeFlag = match.home_flag || '';
  const awayFlag = match.away_flag || '';

  return `
    <div class="match-card" data-state="${status}" data-start-time="${match.start_time}" data-end-time="${match.end_time}">
      <a href="${getMatchUrl(match)}">
        <div class="league-header">
          <span class="league-name" title="${match.league}">${match.league}</span>
          <span class="live-badge">${badgeText}</span>
        </div>

        <div class="match-content">
          <div class="team">
            <div class="team-flag">
              ${homeFlag.includes('http') ? `<img src="${homeFlag}" alt="${match.home_team}">` : homeFlag}
            </div>
            <div class="team-name">${match.home_team}</div>
          </div>

          <div class="match-status">
            <div class="vs-text">VS</div>
            ${statusContent}
          </div>

          <div class="team">
            <div class="team-flag">
              ${awayFlag.includes('http') ? `<img src="${awayFlag}" alt="${match.away_team}">` : awayFlag}
            </div>
            <div class="team-name">${match.away_team}</div>
          </div>
        </div>

        <div class="match-footer">
          ${footerContent}
        </div>
      </a>
    </div>
  `;
}

function sortMatches(matches) {
  return matches.sort((a, b) => {
    const aStatus = getMatchStatus(a);
    const bStatus = getMatchStatus(b);
    const statusPriority = { 'live': 1, 'upcoming': 2, 'ended': 3 };
    if (aStatus !== bStatus) return statusPriority[aStatus] - statusPriority[bStatus];
    const aStart = new Date(a.start_time);
    const bStart = new Date(b.start_time);
    if (aStatus === 'live' || aStatus === 'upcoming') return aStart - bStart;
    return bStart - aStart;
  });
}

function renderMatches(matches) {
  const container = document.getElementById('matches');
  container.innerHTML = '';
  sortMatches(matches.filter(m => getMatchStatus(m) !== 'ended')).forEach(match => {
    container.innerHTML += createMatchCard(match);
  });
}

function updateCountdowns() {
  const matchCards = document.querySelectorAll('.match-card');
  let needsRefresh = false;

  matchCards.forEach((card) => {
    const startTime  = card.getAttribute('data-start-time');
    const endTime    = card.getAttribute('data-end-time');
    const footer     = card.querySelector('.match-footer');
    const matchStatus = card.querySelector('.match-status');

    if (!startTime || !footer || !matchStatus) return;

    const now   = new Date();
    const start = new Date(startTime);
    const end   = new Date(endTime);

    if (now >= start && now < end) {
      if (!matchStatus.querySelector('.live-indicator')) needsRefresh = true;
    } else if (now < start) {
      const timeRemaining = getTimeDifference(startTime);
      if (timeRemaining) {
        // Arrow is now part of the text — include it in the comparison
        const btn = footer.querySelector('.live-score-btn') || footer;
        const newText = `${timeRemaining} \u2192`;
        if (btn.textContent !== newText) btn.textContent = newText;
      } else {
        needsRefresh = true;
      }
    } else if (now >= end) {
      if (!matchStatus.querySelector('.match-time') || matchStatus.querySelector('.match-time').textContent !== 'END') {
        needsRefresh = true;
      }
    }
  });

  if (needsRefresh && !refreshPending) {
    refreshPending = true;
    fetchMatchData();
  }
}

async function fetchMatchData() {
  try {
    const response = await fetch(JSON_URL);
    const matches  = await response.json();
    renderMatches(matches);
    refreshPending = false;
  } catch (error) {
    console.error('Error fetching match data:', error);
    refreshPending = false;
    document.getElementById('matches').innerHTML = '<p style="text-align:center;padding:20px;color:#94a3b8;">Error loading matches. Please try again later.</p>';
  }
}

document.addEventListener('DOMContentLoaded', function () {
  fetchMatchData();
  setInterval(updateCountdowns, 1000);
  setInterval(fetchMatchData,   60000);
});
</script>

      if (s1) s1.textContent = sStr[0];
      if (s2) s2.textContent = sStr[1];

  async function load() {
    if (!idParam) { state.phase = 'notfound'; render(); return; }

    try {
      const res = await fetch(CONFIG.infoJsonUrl + '?t=' + Date.now(), { cache: 'no-store' });
      if (!res.ok) throw new Error('HTTP ' + res.status);
      const info = await res.json();
      const matches = Array.isArray(info) ? info : (info.matches || info.data || []);
      
      const match = matches.find(m => m.id === idParam || m.match_id === idParam);
      
      if (!match) {
        state.phase = 'notfound';
        render();
        return;
      }

      state.match = match;

      const now = Date.now();
      const rawStart = new Date(match.start_time).getTime();
      let rawEnd = match.end_time ? new Date(match.end_time).getTime() : null;
      if (rawEnd !== null && isNaN(rawEnd)) rawEnd = null;

      state.startTs = inflate(rawStart);
      state.endTs   = rawEnd === null ? null : inflate(rawEnd);

      const start = state.startTs;
      const end   = state.endTs;

      if (isNaN(start)) {
        state.phase = 'error'; 
      } else {
        const diffMins = (start - now) / 60000;

        if (now >= start && (!end || now <= end)) {
          state.phase = 'live';
        } else if (end && now > end) {
          state.phase = 'ended';
        } else if (diffMins > 30) {
          state.phase = 'upcoming_early';
        } else {
          state.phase = 'upcoming_countdown';
        }
      }
      render();

    } catch (e) {
      state.phase = 'error';
      render();
    }
  }

      const res = await fetch(CONFIG.liveJsonUrl + '?t=' + Date.now(), { cache: 'no-store' });
      if (!res.ok) throw new Error('HTTP ' + res.status);
      
      const encryptedText = await res.text();
      
      if (!_nativesClean()) { state.streamsPhase = 'error'; render(); return; }

      const rawBytes = _b(encryptedText.trim());
      let salt = '', encryptedBytes = null;
      
      if (rawBytes.length > 16) {
        for (let i = 0; i < 16; i++) salt += B2C[rawBytes[i]];
        encryptedBytes = rawBytes.slice(16);
      } else {
        throw new Error('Invalid manifest format');
      }

      const liveMap = _rt(encryptedBytes, salt);
      
      if (!liveMap) {
         state.streamsPhase = 'empty'; 
         render();
         return;
      }

      const matchId = state.match.id || state.match.match_id;
      const streams = liveMap[matchId];

      if (Array.isArray(streams) && streams.length > 0) {
        state.streams = streams;
        state.streamsPhase = 'ready';
      } else {
        state.streams = [];
        state.streamsPhase = 'empty';
      }
    } catch (e) {
      state.streamsPhase = 'error';
    }
    render();
  };

  // Initial load
  load();

  // Refresh state every 30s
  setInterval(() => {
    if (state.streamsPhase === 'ready' || state.streamsPhase === 'error' || state.streamsPhase === 'empty') {
      if (state.match) {
        const now = Date.now();
        const end = state.endTs;
        if (end && !isNaN(end) && now > end) {
          state.phase = 'ended';
          state.streamsPhase = 'idle';
          if (state.timerId) clearInterval(state.timerId);
          render();
        }
      }
    } else {
      load();
    }
  }, 30000);

})();
</script>
